-- 03_views_procedures_triggers.sql
CREATE TRIGGER trg_price_update
AFTER UPDATE ON Track
FOR EACH ROW
BEGIN
    IF OLD.UnitPrice <> NEW.UnitPrice THEN
        INSERT INTO TrackPrice_Audit (track_id, old_price, new_price, change_date, change_user)
        VALUES (OLD.TrackId, OLD.UnitPrice, NEW.UnitPrice, NOW(), USER());
    END IF;
END $$
DELIMITER ;
